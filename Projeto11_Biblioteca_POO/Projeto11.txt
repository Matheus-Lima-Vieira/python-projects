 class Livro: #TUDO que for livro vai ter as características abaixo
    def __init__(self, titulo, autor, ano): # Método construtor para inicializar os atributos
        self.titulo = titulo
        self.autor = autor
        self.ano = ano

    def descricao(self):    # Método da classe (comportamento)
        return f"{self.titulo} {self.autor}, ano {self.ano}"

meu_livro = Livro("Harry Potter", "J. K. Rowling", 1997)    # Criando um objeto (instância) da classe

print(meu_livro.descricao())    # Usando o método do objeto
